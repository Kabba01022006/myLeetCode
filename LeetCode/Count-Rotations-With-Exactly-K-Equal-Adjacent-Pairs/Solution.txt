1class Solution {
2    public int countRotations(String s, int k) {
3        //loop iterations = length of string
4        int n=s.length();
5        char[] arr = new char[s.length()];
6        for(int i=0;i<s.length();i++){
7            arr[i] = s.charAt(i);
8        }
9
10        int score;
11        int count=0;
12        for(int i=0;i<n;i++){
13            rotate(arr);
14            score=0;
15            for(int j=0;j<n-1;j++){
16                if(arr[j]==arr[j+1]){
17                    score++;
18                }
19            }
20            if(score==k){
21                count++;
22            }
23        }
24        return count;
25    }
26    public void rotate(char[] array){
27        int len = array.length;
28        int start=0;
29        int end=len-2;
30        while(start<=end){
31            char temp = array[start];
32            array[start] = array[end];
33            array[end] = temp;
34            start++;
35            end--;
36        }
37        start=0;
38        end=len-1;
39        while(start<=end){
40            char temp = array[start];
41            array[start] = array[end];
42            array[end] = temp;
43            start++;
44            end--;
45        }
46    }
47}
